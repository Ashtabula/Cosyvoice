def percentile(a,q):
 a=sorted(a);p=(len(a)-1)*q;i=int(p);return a[i]+(a[min(i+1,len(a)-1)]-a[i])*(p-i)
def summary(a):return {'count':len(a),'min':min(a),'median':statistics.median(a),'mean':statistics.mean(a),'p90':percentile(a,.9),'p95':percentile(a,.95),'p99':percentile(a,.99),'max':max(a),'sum':sum(a)}
def regression(xs,ys):
 xm,ym=statistics.mean(xs),statistics.mean(ys);xx=sum((x-xm)**2 for x in xs);yy=sum((y-ym)**2 for y in ys);xy=sum((x-xm)*(y-ym) for x,y in zip(xs,ys));return {'slopeMsPerContextToken':xy/xx,'pearsonR':xy/math.sqrt(xx*yy),'meaning':'descriptive; sequential temporal/clock/thermal correlation, not causal attention growth or independent p-value'}
